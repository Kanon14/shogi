# Shogi V1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local two-player browser shogi game with a tested TypeScript rules engine and a playable React UI.

**Architecture:** The rules engine lives in pure TypeScript under `src/game` and does not import React. React components under `src/components` render the board, hands, status, promotion prompt, and move history while delegating all legality and state transitions to the game layer.

**Tech Stack:** Vite, React, TypeScript, Vitest, CSS.

**Spec:** `docs/superpowers/specs/2026-08-28-shogi-v1-design.md`

## Global Constraints

- Local two-player play only.
- No online multiplayer.
- No computer opponent.
- No account system or persistence.
- No puzzle or training mode.
- Standard 9x9 shogi rules.
- The rules engine must not import React.
- Implement milestone-by-milestone; after each milestone, verify and stop for user review.

Create `src/game/initialPosition.ts`:

```ts
import type { Board, GameState, Piece, Square } from './types';

const emptyBoard = (): Board => Array.from({ length: 9 }, () => Array<Piece | null>(9).fill(null));

const toIndex = (square: Square) => ({
  row: square.rank - 1,
  col: 9 - square.file,
});

const place = (board: Board, square: Square, piece: Piece) => {
  const { row, col } = toIndex(square);
  board[row][col] = piece;
};

export const getPieceAt = (board: Board, square: Square): Piece | null => {
  const { row, col } = toIndex(square);
  return board[row]?.[col] ?? null;
};

export const createInitialGameState = (): GameState => {
  const board = emptyBoard();

  const backRank = ['lance', 'knight', 'silver', 'gold', 'king', 'gold', 'silver', 'knight', 'lance'] as const;
  backRank.forEach((kind, index) => {
    place(board, { file: 9 - index, rank: 9 }, { owner: 'sente', kind });
    place(board, { file: 9 - index, rank: 1 }, { owner: 'gote', kind });
  });

  place(board, { file: 8, rank: 8 }, { owner: 'sente', kind: 'bishop' });
  place(board, { file: 2, rank: 8 }, { owner: 'sente', kind: 'rook' });
  place(board, { file: 2, rank: 2 }, { owner: 'gote', kind: 'bishop' });
  place(board, { file: 8, rank: 2 }, { owner: 'gote', kind: 'rook' });

  for (let file = 1; file <= 9; file += 1) {
    place(board, { file, rank: 7 }, { owner: 'sente', kind: 'pawn' });
    place(board, { file, rank: 3 }, { owner: 'gote', kind: 'pawn' });
  }

  return {
    board,
    hands: { sente: {}, gote: {} },
    currentPlayer: 'sente',
    history: [],
    status: { type: 'playing' },
  };
};
```

Create `src/game/state.ts` with immutable helpers and the complete transition shape:

```ts
import { formatMove } from './notation';
import { isCheckmate, isInCheck } from './legalMoves';
import type { AnyPieceKind, Board, GameMove, GameState, Hand, Hands, PieceKind, Player } from './types';

export const promote = (kind: AnyPieceKind): AnyPieceKind => ({
  rook: 'dragon',
  bishop: 'horse',
  silver: 'promotedSilver',
  knight: 'promotedKnight',
  lance: 'promotedLance',
  pawn: 'tokin',
}[kind] ?? kind) as AnyPieceKind;

export const demote = (kind: AnyPieceKind): PieceKind => ({
  dragon: 'rook',
  horse: 'bishop',
  promotedSilver: 'silver',
  promotedKnight: 'knight',
  promotedLance: 'lance',
  tokin: 'pawn',
}[kind] ?? kind) as PieceKind;

const opponentOf = (player: Player): Player => (player === 'sente' ? 'gote' : 'sente');

const cloneBoard = (board: Board): Board => board.map((row) => row.map((piece) => (piece ? { ...piece } : null)));

const cloneHands = (hands: Hands): Hands => ({
  sente: { ...hands.sente },
  gote: { ...hands.gote },
});

const addToHand = (hand: Hand, kind: PieceKind): Hand => ({
  ...hand,
  [kind]: (hand[kind] ?? 0) + 1,
});

const removeFromHand = (hand: Hand, kind: PieceKind): Hand => {
  const current = hand[kind] ?? 0;
  if (current <= 1) {
    const next = { ...hand };
    delete next[kind];
    return next;
  }
  return { ...hand, [kind]: current - 1 };
};

const statusAfterMove = (state: GameState, nextPlayer: Player) => {
  const mover = opponentOf(nextPlayer);
  if (isCheckmate(state, nextPlayer)) return { type: 'checkmate' as const, winner: mover, loser: nextPlayer };
  if (isInCheck(state, nextPlayer)) return { type: 'check' as const, checkedPlayer: nextPlayer };
  return { type: 'playing' as const };
};

export const applyMove = (state: GameState, move: GameMove): GameState => {
  const board = cloneBoard(state.board);
  const hands = cloneHands(state.hands);
  const nextPlayer = opponentOf(state.currentPlayer);
  const label = formatMove(state, move);

  if (move.type === 'move') {
    const fromRow = move.from.rank - 1;
    const fromCol = 9 - move.from.file;
    const toRow = move.to.rank - 1;
    const toCol = 9 - move.to.file;
    const piece = board[fromRow][fromCol];
    if (!piece) throw new Error('Cannot move from an empty square.');

    const captured = board[toRow][toCol];
    if (captured) {
      hands[state.currentPlayer] = addToHand(hands[state.currentPlayer], demote(captured.kind));
    }

    board[fromRow][fromCol] = null;
    board[toRow][toCol] = { ...piece, kind: move.promote ? promote(piece.kind) : piece.kind };
  } else {
    const toRow = move.to.rank - 1;
    const toCol = 9 - move.to.file;
    if (board[toRow][toCol]) throw new Error('Cannot drop onto an occupied square.');
    if ((hands[state.currentPlayer][move.pieceKind] ?? 0) <= 0) throw new Error('Cannot drop a piece that is not in hand.');

    hands[state.currentPlayer] = removeFromHand(hands[state.currentPlayer], move.pieceKind);
    board[toRow][toCol] = { owner: state.currentPlayer, kind: move.pieceKind };
  }

  const nextState = {
    ...state,
    board,
    hands,
    currentPlayer: nextPlayer,
    history: [...state.history, { id: state.history.length + 1, player: state.currentPlayer, label }],
  };

  return {
    ...nextState,
    status: statusAfterMove(nextState, nextPlayer),
  };
};
```

  const pieceName = piece?.kind ?? 'piece';
  const suffix = move.promote ? '+' : '';
  return `${state.currentPlayer} ${pieceName}${suffix} ${move.from.file}${move.from.rank}-${move.to.file}${move.to.rank}`;
};

- [ ] **Step 6: Implement legal drops and endgame detection**

In `src/game/legalMoves.ts`, implement legal drops and checkmate with these decisions:

```ts
export const getLegalDropMoves = (state: GameState, pieceKind: PieceKind): DropMove[] => {
  const count = state.hands[state.currentPlayer][pieceKind] ?? 0;
  if (count <= 0) return [];

  const candidates: DropMove[] = [];
  for (let rank = 1; rank <= 9; rank += 1) {
    for (let file = 1; file <= 9; file += 1) {
      const to = { file, rank };
      if (state.board[rank - 1][9 - file]) continue;
      if (isDeadZoneDrop(pieceKind, state.currentPlayer, to)) continue;
      if (pieceKind === 'pawn' && fileHasUnpromotedPawn(state, state.currentPlayer, file)) continue;

      const move: DropMove = { type: 'drop', pieceKind, to };
      const next = applyDropForSearch(state, move);
      if (isInCheck(next, state.currentPlayer)) continue;
      if (pieceKind === 'pawn' && isCheckmate(next, opponentOf(state.currentPlayer))) continue;

      candidates.push(move);
    }
  }

  return candidates;
};

export const isCheckmate = (state: GameState, player: Player): boolean => {
  if (!isInCheck(state, player)) return false;
  const searchState = { ...state, currentPlayer: player };

  for (let rank = 1; rank <= 9; rank += 1) {
    for (let file = 1; file <= 9; file += 1) {
      const piece = searchState.board[rank - 1][9 - file];
      if (piece?.owner === player && getLegalBoardMoves(searchState, { file, rank }).length > 0) {
        return false;
      }
    }
  }

  return Object.entries(searchState.hands[player]).every(([kind, count]) => {
    if (!count) return true;
    return getLegalDropMoves(searchState, kind as PieceKind).length === 0;
  });
};
```

Support helpers in the same file:

- `isDeadZoneDrop(kind, player, to)` rejects pawn/lance on the last rank and knight on the last two ranks.
- `fileHasUnpromotedPawn(state, player, file)` scans the file for an owned unpromoted pawn.
- `applyDropForSearch(state, move)` clones the board and hands, places the dropped piece, removes one hand count, and keeps history unchanged.
- `opponentOf(player)` returns `sente` for `gote` and `gote` for `sente`.
